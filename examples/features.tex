\section{定理与引用}
\begin{Theorem}\label{thm:positive}
若 $a>0$ 且 $b>0$，则 $a+b>0$。
\end{Theorem}
\begin{Lemma}\label{lem:square}
对任意实数 $x$，有 $x^2\geq0$。
\end{Lemma}
\begin{Corollary}\label{cor:square}
对任意实数 $x$，有 $x^2+1>0$。
\end{Corollary}
\begin{Proposition}\label{prop:zero}
若 $x=0$，则 $x^2=0$。
\end{Proposition}
\begin{Definition}\label{def:even}
能被 2 整除的整数称为偶数。
\end{Definition}
\begin{Example}\label{ex:even}
整数 4 是偶数。
\end{Example}
同一节内共用编号，但引用名称分别保留：
\autoref{thm:positive}、\autoref{lem:square}、\autoref{cor:square}、
\autoref{prop:zero}、\autoref{def:even}、\autoref{ex:even}。
